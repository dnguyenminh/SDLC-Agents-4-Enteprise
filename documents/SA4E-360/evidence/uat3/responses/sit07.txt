1. writable<T>() — Basic Reactive Store
export function writable<T>(initialValue: T) {
  let value = initialValue;
  const subscribers: Array<(v: T) => void> = [];

  return {
    set(v: T) { value = v; subscribers.forEach((s) => s(value)); },
    update(fn: (v: T) => T) { value = fn(value); subscribers.forEach((s) => s(value)); },
    subscribe(run: (v: T) => void) {
      run(value);
      subscribers.push(run);
      return () => {
        const i = subscribers.indexOf(run);
        if (i !== -1) subscribers.splice(i, 1);
      };
    },
    get value() { return value; },
  } as any;
}CopyApplyInserttypescript

set() — Thay đổi value, notify tất cả subscribersupdate() — Transform value, notify subscriberssubscribe() — Subscribe + auto-unsubscribevalue — Direct access (debugging)
